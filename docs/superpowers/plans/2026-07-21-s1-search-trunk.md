```js
#!/usr/bin/env node
/**
 * Golden-query relevance harness (S1, spec §6.8). Measures precision@5 for a
 * fixed query set against the LIVE Meilisearch index, replicating the app's
 * read path (same filter baseline as buildFilter's invariant clause).
 *
 *   node scripts/golden-queries.mjs                 # default engine behavior
 *   node scripts/golden-queries.mjs --titles-only   # precision pass (attributesToSearchOn)
 *   node scripts/golden-queries.mjs --save          # also write scripts/golden-baseline.json
 */
import { readFileSync, writeFileSync } from "node:fs";
import { dirname, join } from "node:path";
import { fileURLToPath } from "node:url";

const here = dirname(fileURLToPath(import.meta.url));

function loadEnvLocal() {
  try {
    const text = readFileSync(join(here, "..", ".env.local"), "utf8");
    for (const line of text.split("\n")) {
      const m = line.match(/^\s*([A-Z0-9_]+)\s*=\s*"?([^"\n]*)"?\s*$/);
      if (m && !(m[1] in process.env)) process.env[m[1]] = m[2];
    }
  } catch {
    /* no .env.local — rely on ambient env */
  }
}
loadEnvLocal();

const HOST = (process.env.MEILISEARCH_HOST ?? process.env.MEILI_URL ?? "").replace(/\/+$/, "");
const KEY = process.env.MEILISEARCH_SEARCH_KEY ?? process.env.MEILISEARCH_API_KEY ?? process.env.MEILI_KEY ?? "";
if (!HOST || !KEY) {
  console.error("Missing MEILISEARCH_HOST / MEILISEARCH_SEARCH_KEY (or MEILI_URL / MEILI_KEY).");
  process.exit(1);
}

const titlesOnly = process.argv.includes("--titles-only");
const save = process.argv.includes("--save");
const { queries } = JSON.parse(readFileSync(join(here, "golden-queries.json"), "utf8"));

async function searchTop5(q) {
  const body = {
    q,
    filter: ['vacancy_status != "closed"'],
    hitsPerPage: 5,
    page: 1,
    ...(titlesOnly ? { attributesToSearchOn: ["title", "company"] } : {}),
  };
  const res = await fetch(`${HOST}/indexes/jobs/search`, {
    method: "POST",
    headers: { "content-type": "application/json", authorization: `Bearer ${KEY}` },
    body: JSON.stringify(body),
  });
  if (!res.ok) throw new Error(`search "${q}" failed: ${res.status} ${await res.text()}`);
  const json = await res.json();
  return (json.hits ?? []).map((h) => ({ title: h.title ?? "", company: h.company ?? "" }));
}

const perQuery = [];
for (const spec of queries) {
  const hits = await searchTop5(spec.q);
  const re = new RegExp(spec.expectCompany ?? spec.expect, "i");
  const field = spec.expectCompany ? "company" : "title";
  const matched = hits.filter((h) => re.test(h[field])).length;
  const precisionAt5 = hits.length ? matched / 5 : 0;
  perQuery.push({ q: spec.q, precisionAt5, top5: hits.map((h) => `${h.title} | ${h.company}`) });
  console.log(`${precisionAt5.toFixed(1).padStart(4)}  ${spec.q}`);
}
const mean = perQuery.reduce((a, r) => a + r.precisionAt5, 0) / perQuery.length;
console.log(`\nmode: ${titlesOnly ? "titles-only" : "default"}  mean precision@5: ${mean.toFixed(3)}`);

if (save) {
  writeFileSync(
    join(here, "golden-baseline.json"),
    JSON.stringify({ generatedAt: new Date().toISOString(), mode: titlesOnly ? "titles-only" : "default", mean, perQuery }, null, 2) + "\n",
  );
  console.log("wrote scripts/golden-baseline.json");
}
```

```ts
/**
 * Deterministic query understanding for jobs search (S1, spec §6.2).
 *
 * Work-arrangement tokens typed in the query become the SAME strict
 * `work_type` facet filter the FilterPanel applies when clicked — "typing it
 * = clicking it". A fixed phrase table, no NLP, so it can never misfire on
 * novel input.
 *
 * Seniority tokens are deliberately NOT parsed into `experience_level`
 * filters: measured live 2026-07-21, the corpus is 76% "mid" (classifier
 * dump bucket), so a seniority filter would DROP correctly-titled senior
 * roles that title-first ranking (see searchMeili's precision pass) already
 * orders correctly.
 *
 * Pure module — no I/O — so behavior locks under unit test.
 */
import type { JobSearchParams } from "./types";

export interface ParsedQuery {
  /** Query text with recognized tokens removed (may be empty). */
  residualQ: string;
  /** Canonical `work_type` facet values implied by the query (corpus vocabulary: remote | hybrid | onsite). */
  workTypes: string[];
}

/** Longest phrases first, so "work from home" is consumed before any single-word pass could see its words. */
const WORK_TYPE_PHRASES: ReadonlyArray<readonly [RegExp, string]> = [
  [/\bwork[- ]from[- ]home\b/gi, "remote"],
  [/\bin[- ]office\b/gi, "onsite"],
  [/\bon[- ]site\b/gi, "onsite"],
  [/\bwfh\b/gi, "remote"],
  [/\bremote\b/gi, "remote"],
  [/\bhybrid\b/gi, "hybrid"],
  [/\bonsite\b/gi, "onsite"],
];

export function parseWorkTypeTokens(q: string): ParsedQuery {
  let residual = q;
  const found = new Set<string>();
  for (const [re, canonical] of WORK_TYPE_PHRASES) {
    const next = residual.replace(re, " ");
    if (next !== residual) {
      found.add(canonical);
      residual = next;
    }
  }
  return { residualQ: residual.replace(/\s+/g, " ").trim(), workTypes: [...found] };
}

/**
 * Rewrite search params with parsed work-type tokens applied as facets.
 * Explicit UI facet selection always wins: if the user already narrowed
 * `work_type` in the FilterPanel, the typed token is left in the query
 * untouched rather than second-guessing their click.
 */
export function applyQueryUnderstanding(p: JobSearchParams): JobSearchParams {
  const q = (p.q ?? "").trim();
  if (!q) return p;
  const { residualQ, workTypes } = parseWorkTypeTokens(q);
  if (workTypes.length === 0) return p;
  if (p.facets?.work_type?.length) return p;
  return { ...p, q: residualQ, facets: { ...p.facets, work_type: workTypes } };
}
```

```ts
export const PRECISION_SEARCH_ATTRIBUTES = ["title", "company"] as const;
/** Pass-1 result floor below which the wide pass runs. Spec §6.1 default. */
export const WIDEN_THRESHOLD = 10;

/** Pure widen policy: only a real query can widen (no-query browses are already wide). */
export function shouldWidenSearch(pass1Total: number, q: string): boolean {
```

Then restructure `searchMeili` so the existing legacy-400-retry dance becomes a reusable single-pass runner, and the two-pass policy sits on top. Replace the body of `searchMeili` (keep the unconfigured early-return exactly as is):

```ts
/** Read-path: search the jobs index. Two-pass (spec §6.1): a title/company
 *  precision pass first; if it comes back thin for a real query, one wide
 *  (description-included) pass, flagged `widened` so the UI can say so. */
export async function searchMeili(p: JobSearchParams): Promise<JobSearchResult> {
  const h = host();
  if (!h) {
    return {
      hits: [],
      total: 0,
      backend: "meilisearch",
      configured: false,
      note: "MEILISEARCH_HOST not set",
    };
  }

  const runPass = async (attributesToSearchOn?: readonly string[]): Promise<JobSearchResult> => {
    const post = (legacy: boolean) =>
      meiliFetch(`${h}/indexes/${JOBS_INDEX}/search`, {
        method: "POST",
        headers: { "content-type": "application/json", authorization: `Bearer ${searchKey()}` },
        cache: "no-store",
        body: JSON.stringify(buildSearchBody(p, legacy, attributesToSearchOn)),
      });

    let res = await post(false);
    if (!res.ok && res.status === 400) {
      const text = await res.text().catch(() => "");
      // TRANSITIONAL: posted_ts isn't sortable/filterable on the live index yet
      // (the settings PATCH + reindex run on the sync box, separately from this
      // deploy). Retry once without it instead of 400ing every search. Safe to
      // delete this branch once that reindex has landed.
      if (/posted_ts/i.test(text)) {
        console.warn("posted_ts not yet indexed — falling back to legacy sort/filter");
        res = await post(true);
      } else {
        throw new Error(`Meilisearch search failed: ${res.status} ${text.slice(0, 300)}`);
      }
    }
    if (!res.ok) {
      const text = await res.text().catch(() => "");
      throw new Error(`Meilisearch search failed: ${res.status} ${text.slice(0, 300)}`);
    }
    const body = (await res.json()) as {
      hits: PublicJob[];
      estimatedTotalHits?: number;
      totalHits?: number;
      facetDistribution?: Record<string, Record<string, number>>;
    };
    return {
      hits: body.hits ?? [],
      total: body.totalHits ?? body.estimatedTotalHits ?? body.hits?.length ?? 0,
      backend: "meilisearch",
      configured: true,
      facetDistribution: body.facetDistribution as FacetDistribution | undefined,
    };
  };

  const q = (p.q ?? "").trim();
  if (!q) return runPass(); // browse mode: already wide, single pass
  const precise = await runPass(PRECISION_SEARCH_ATTRIBUTES);
  if (!shouldWidenSearch(precise.total, q)) return precise;
  const wide = await runPass();
  // The widened flag is honest signal, not decoration: only claim it when
  // widening actually surfaced more than the precision pass had.
  return wide.total > precise.total ? { ...wide, widened: true } : precise;
}
```

```js
#!/usr/bin/env node
/**
 * Push ONLY the synonyms setting to the live jobs index. Deliberately narrow:
 * a full settings PATCH would add "id" to filterableAttributes and trigger a
 * multi-hour reindex of ~1.8M docs — that re-provision belongs to the box's
 * sync worker, not this script.
 *
 *   node scripts/apply-synonyms.mjs             # apply JOBS_SYNONYMS
 *   node scripts/apply-synonyms.mjs --rollback  # clear all synonyms
 *
 * Env: MEILISEARCH_HOST + MEILISEARCH_ADMIN_KEY (or box names MEILI_URL /
 * MEILI_ADMIN_KEY). Reads .env.local; admin key is usually NOT there — export
 * it, or run this on the box.
 */
import { readFileSync } from "node:fs";
import { dirname, join } from "node:path";
import { fileURLToPath } from "node:url";

const here = dirname(fileURLToPath(import.meta.url));
try {
  for (const line of readFileSync(join(here, "..", ".env.local"), "utf8").split("\n")) {
    const m = line.match(/^\s*([A-Z0-9_]+)\s*=\s*"?([^"\n]*)"?\s*$/);
    if (m && !(m[1] in process.env)) process.env[m[1]] = m[2];
  }
} catch {}

const HOST = (process.env.MEILISEARCH_HOST ?? process.env.MEILI_URL ?? "").replace(/\/+$/, "");
const KEY = process.env.MEILISEARCH_ADMIN_KEY ?? process.env.MEILI_ADMIN_KEY ?? "";
if (!HOST || !KEY) {
  console.error("Missing MEILISEARCH_HOST/MEILI_URL or MEILISEARCH_ADMIN_KEY/MEILI_ADMIN_KEY.");
  process.exit(1);
}

// Inline copy of JOBS_SYNONYMS (this is an .mjs script; it can't import the .ts
// module). MUST match src/lib/jobs/meilisearch.ts JOBS_SYNONYMS — that export
// is the source of truth; update both together.
const JOBS_SYNONYMS = {
  swe: ["software engineer"],
  "software engineer": ["swe"],
  sre: ["site reliability engineer"],
  "site reliability engineer": ["sre"],
  pm: ["product manager", "project manager"],
  k8s: ["kubernetes"],
  kubernetes: ["k8s"],
  ml: ["machine learning"],
  "machine learning": ["ml"],
  ux: ["user experience"],
  qa: ["quality assurance"],
  hr: ["human resources"],
  devops: ["dev ops"],
};

const rollback = process.argv.includes("--rollback");
const payload = rollback ? {} : JOBS_SYNONYMS;

const res = await fetch(`${HOST}/indexes/jobs/settings/synonyms`, {
  method: "PUT",
  headers: { "content-type": "application/json", authorization: `Bearer ${KEY}` },
  body: JSON.stringify(payload),
});
if (!res.ok) {
  console.error(`PUT synonyms failed: ${res.status} ${await res.text()}`);
  process.exit(1);
}
const { taskUid } = await res.json();
for (;;) {
  const t = await (await fetch(`${HOST}/tasks/${taskUid}`, { headers: { authorization: `Bearer ${KEY}` } })).json();
  if (t.status === "succeeded") break;
  if (t.status === "failed") {
    console.error("task failed:", JSON.stringify(t.error));
    process.exit(1);
  }
  await new Promise((r) => setTimeout(r, 500));
}
const current = await (
  await fetch(`${HOST}/indexes/jobs/settings/synonyms`, { headers: { authorization: `Bearer ${KEY}` } })
).json();
console.log(rollback ? "synonyms cleared." : "synonyms applied:", JSON.stringify(current, null, 2));
```
